// 1. Limpia las variables existentes en el espacio de trabajo
clear;

// 2. Cierra ventanas gráficas activas
clf;

// 3. Define el periodo fundamental N0
N0 = 8;

// 4. Calcula la frecuencia angular discreta fundamental Omega0 = 2*pi/8
Omega0 = (2 * %pi) / N0;

// 5. Define el vector de tiempo discreto para un periodo n = 0 a 7
n = 0:(N0 - 1);

// 6. Define el vector de armónicos k desde 0 hasta N0-1
k = 0:(N0 - 1);

// 7. Inicializa y construye la primera señal u1 en un periodo
u1 = zeros(1, N0);
u1(n >= 0 & n <= 4) = 1;

// 8. Construye la segunda señal senoidal u2 en un periodo
u2 = 2 * sin(Omega0 * n);

// 9. Construye la matriz base de análisis DTFS de dimension (N0 x N0) con exponente -j*k*Omega0*n
//    k' genera un vector columna y n un vector fila para producir el producto exterior
W = exp(-%i * Omega0 * (k' * n));

// 10. Calcula los coeficientes a_k de u1 aplicando la definición matricial: (1/N0) * W * u1'
a1 = (1 / N0) * (W * u1');

// 11. Calcula los coeficientes a_k de u2 aplicando la definición matricial: (1/N0) * W * u2'
a2 = (1 / N0) * (W * u2');

// 12. Imprime en la consola los valores numéricos obtenidos para a1
disp("Coeficientes a_k de u1[n]:");
disp(a1);

// 13. Imprime en la consola los valores numéricos obtenidos para a2
disp("Coeficientes a_k de u2[n]:");
disp(a2);

// 14. Divide la ventana gráfica en 2 filas y 2 columnas, seleccionando el cuadrante 1
subplot(2, 2, 1);
// 15. Grafica la magnitud |a1_k|
plot2d3(k, abs(a1));
plot(k, abs(a1), 'bo');
xgrid();
xlabel('k');
ylabel('|a_{1,k}|');
title('Magnitud de a_k para u_1[n]');

// 16. Selecciona el cuadrante 2
subplot(2, 2, 2);
// 17. Grafica la fase en radianes de a1_k
plot2d3(k, atan(imag(a1), real(a1)));
plot(k, atan(imag(a1), real(a1)), 'bo');
xgrid();
xlabel('k');
ylabel('Fase (rad)');
title('Fase de a_k para u_1[n]');

// 18. Selecciona el cuadrante 3
subplot(2, 2, 3);
// 19. Grafica la magnitud |a2_k|
plot2d3(k, abs(a2));
plot(k, abs(a2), 'ro');
xgrid();
xlabel('k');
ylabel('|a_{2,k}|');
title('Magnitud de a_k para u_2[n]');

// 20. Selecciona el cuadrante 4
subplot(2, 2, 4);
// 21. Grafica la fase en radianes de a2_k
plot2d3(k, atan(imag(a2), real(a2)));
plot(k, atan(imag(a2), real(a2)), 'ro');
xgrid();
xlabel('k');
ylabel('Fase (rad)');
title('Fase de a_k para u_2[n]');
